from http.server import BaseHTTPRequestHandler, HTTPServer from datetime import datetime class Handler(BaseHTTPRequestHandler): def do_GET(self): ip = self.client_address[0] time = datetime.now().strftime("%Y-%m-%d %H:%M:%S") print(f"[{time}] Visitor IP: {ip}") self.send_response(200) self.send_header("Content-type", "text/html") self.end_headers() page = """ Test Page

IP Testing Page

This page is for testing purposes.

""" self.wfile.write(page.encode()) server = HTTPServer(("0.0.0.0", 8080), Handler) print("Server running on port 8080...") server.serve_forever()
